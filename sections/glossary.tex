{
	name={inaccessible document},
	description={a \latexspaced document that has intentionally been made inaccessible for screen reader users.}
}

{
	name={right-tailed test},
	description={a test for whether the hypothesis should be rejected \parencite{right-tailed_test}.}
}

{
	name={accessible document},

}

\newglossaryentry{metadata}
{
	name={metadata},
	description={data about data, such as a file's structure.}
}

\newglossaryentry{alt_text}
{
	name={alternative text},
	description={text used in place of non-text components for accessibility purposes.}
}

\newglossaryentry{likert}
{
	name={likert scale datum},
	plural={likert scale data},
	description={a scale, usually with 5 or 7 points, indicating a participant's agreement or disagreement with some topic \parencite{likert}.}
}

\newglossaryentry{quan_data}
{
	name={quantitative data},
	description={data that can be represented in numerical form \parencite{quan_data}.}
}

\newglossaryentry{qual_data}
{
	name={qualitative data},
	description={data that cannot be represented numerically, such as descriptions, statements or quotes \parencite{qual_data}.}
}

\newglossaryentry{think_aloud}
{
	name={think-aloud session},
	description={users think aloud while performing a given task, with the goal of identifying areas of improvement in the usability of the given context.}
}

\newglossaryentry{artifact}
{
	name={artifact},
	description={an indicator to a screen reader to skip an image in its reading}
}
